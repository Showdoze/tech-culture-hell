############################################################
# The Franco-British Union - Hegemon Creation Decision
# Commissioned by: OttomaticallyAwesome
#
# Requirement: hold Britannia OR France (no claim needed).
# Effect: create the hegemon "Franco-British Union"; you
# gain a pressed claim on the other empire (so you can war
# for it). Empires you hold become de jure parts.
############################################################

form_franco_british_union = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}
	decision_group_type = major
	desc = form_franco_british_union_desc
	ai_goal = yes
	ai_potential = { always = yes }
	ai_will_do = { base = 100 }
	is_shown = {
		is_independent_ruler = yes
		primary_title.tier >= tier_kingdom
		OR = {
			has_title = title:e_britannia
			has_title = title:e_france
		}
		NOT = { exists = title:h_franco_british_union.holder }
	}
	is_valid = {
		is_imprisoned = no
		# Holding either empire is enough. Forming the Union grants a
		# pressed claim on the other empire (see the effect below).
		custom_description = {
			text = form_franco_british_union_requirement
			OR = {
				has_title = title:e_britannia
				has_title = title:e_france
			}
		}
	}
	effect = {
		# Create the hegemon title for the founder
		title:h_franco_british_union = {
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change_fbu
				add_claim_on_loss = no
			}
			change_title_holder = {
				holder = root
				change = scope:change_fbu
			}
			resolve_title_and_vassal_change = scope:change_fbu
			hidden_effect = {
				copy_title_history = root.primary_title
			}
		}
		# Empires the founder holds become de jure parts of the union
		every_held_title = {
			limit = {
				OR = {
					this = title:e_britannia
					this = title:e_france
				}
			}
			set_de_jure_liege_title = title:h_franco_british_union
		}
		# For the empire the founder does NOT hold:
		#  - if its holder is close family, they become a vassal peacefully
		#  - otherwise the founder gains a pressed claim to war for it
		if = {
			limit = { NOT = { has_title = title:e_britannia } }
			if = {
				limit = {
					exists = title:e_britannia.holder
					title:e_britannia.holder = { is_close_family_of = root }
				}
				title:e_britannia.holder = {
					create_title_and_vassal_change = {
						type = vassal
						save_scope_as = change_brit
						add_claim_on_loss = no
					}
					change_liege = {
						liege = root
						change = scope:change_brit
					}
					resolve_title_and_vassal_change = scope:change_brit
				}
			}
			else = {
				add_pressed_claim = title:e_britannia
			}
			# Empires are often not formed, so also grant a claim on the
			# kingdom of England (unless the founder already holds it).
			if = {
				limit = {
					exists = title:k_england
					NOT = { has_title = title:k_england }
				}
				add_pressed_claim = title:k_england
			}
		}
		if = {
			limit = { NOT = { has_title = title:e_france } }
			if = {
				limit = {
					exists = title:e_france.holder
					title:e_france.holder = { is_close_family_of = root }
				}
				title:e_france.holder = {
					create_title_and_vassal_change = {
						type = vassal
						save_scope_as = change_fra
						add_claim_on_loss = no
					}
					change_liege = {
						liege = root
						change = scope:change_fra
					}
					resolve_title_and_vassal_change = scope:change_fra
				}
			}
			else = {
				add_pressed_claim = title:e_france
			}
			# Also grant a claim on the kingdom of France.
			if = {
				limit = {
					exists = title:k_france
					NOT = { has_title = title:k_france }
				}
				add_pressed_claim = title:k_france
			}
		}
		add_legitimacy = 1500
		add_prestige = 5000
		dynasty = { add_dynasty_prestige = 2500 }
	}
	cost = { gold = 2500 }
}

iron_islands_decision = {
    picture =  { reference = "gfx/interface/illustrations/decisions/decision_realm.dds" }
	decision_group_type = major
	ai_check_interval = 0
	desc = iron_islands_decision_decision_desc
	is_shown = {
		culture = culture:ironborn
        is_independent_ruler = yes
        NOT = {
            exists = title:k_the_iron_islands.holder
        }
	}
    is_valid_showing_failures_only = {
		is_capable_adult = yes
		is_imprisoned = no
		is_independent_ruler = yes
	}
	is_valid = {
		AND = {
			completely_controls = title:d_cyrenaica
			completely_controls = title:d_alexandria
            completely_controls = title:d_delta
            completely_controls = title:d_cairo
            completely_controls = title:d_sinai
            completely_controls = title:d_palestine
            completely_controls = title:d_urdunn
            completely_controls = title:d_eastern_desert
		}
	}
	cost = {
		gold = 500
		prestige = 1500
		piety = 200
	}
	effect = {
		create_title_and_vassal_change = change
		scope:change = {
			set_title_and_vassal_change_type = created
		}
		title:k_the_iron_islands = {
			change_title_holder = {
				holder = root
				change = scope:change
			}
			hidden_effect = {
				set_title_name = k_the_iron_islands
			}
		}
		resolve_title_and_vassal_change = scope:change
		hidden_effect = {
			save_scope_as = founder
			primary_title = {
				save_scope_as = old_title
			}
			create_dynamic_title = {
				tier = empire
				name = "k_the_iron_islands"
			}
			create_title_and_vassal_change = change
			scope:change = {
				set_title_and_vassal_change_type = created
			}
			scope:new_title = {
				change_title_holder = {
					holder = root
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
			every_held_title = {
				limit = {
					tier = tier_kingdom
				}				
				set_de_jure_liege_title = scope:new_title
			}
			every_sub_realm_county = {
				limit = {
					exists = kingdom
					NOT = { exists = kingdom.holder }
					holder.top_liege = root
					kingdom = {
						save_temporary_scope_as = test_kingdom
					}
					holder.top_liege = {
						completely_controls = scope:test_kingdom
					}
				}
				if = {
					limit = {
						NOT = {
							kingdom = {
								is_in_list = additional_de_jure_kingdoms
							}
						}
					}
					kingdom = {
						set_de_jure_liege_title = scope:new_title
						add_to_list = additional_de_jure_kingdoms
					}
				}
			}
			scope:new_title = {
				set_title_name = title:k_the_iron_islands
				set_coa = title:k_the_iron_islands
				set_color_from_title = title:k_the_iron_islands
				set_capital_county = title:k_the_iron_islands.title_capital_county
			}
			set_primary_title_to = scope:new_title
			title:d_cyrenaica = { set_de_jure_liege_title = title:k_the_iron_islands }
			title:d_alexandria = { set_de_jure_liege_title = title:k_the_iron_islands }
		}
		every_held_title = {
			limit = {
				NOR = {
					this = root.primary_title
					this = title:k_the_iron_islands
				}				
                OR = {
                    tier = tier_kingdom
                    tier = tier_empire
                }
            }
            root = { destroy_title = prev }
        }
	}
}
