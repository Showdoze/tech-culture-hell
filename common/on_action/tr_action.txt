## MODDED HOOKED BELOW

tr_satanism_conversion = {
    effect = {
        trigger_event = tr_satanism.01
    }
}

tr_satanism_start = {
    effect = {
        if = {
            limit = {
                faith = faith:satanism
            }
            #set_piety_level = 0
            add_piety = -200
        }
    }
}

tr_hygene = {
    effect = {
        trigger_event = tr_hygene.01
    }
}

tr_hollowing = {
    effect = {
        trigger_event = ds1_event.01
    }
}

tr_birth = {
    effect = {
        trigger_event = tr_birth.01 #emperor child trait
        trigger_event = tr_birth.02 #birthsign trait

        trigger_event = tr_birth.031 #born blind trait
        trigger_event = tr_birth.032 #born deaf trait
        trigger_event = tr_birth.033 #born mute trait
    }
}

tr_sayan_check = {
    effect = { tr_play_sayan_music = yes }
}

tr_artifact_start_spawn = {
    effect = {
        trigger_event = tr_artifact_spawn.01
    }
}

tr_sayan_boost = {
    effect = {
        trigger_event = zenkai_boost.0001
    }
}

tr_traits_start = {
    effect = {
        every_living_character = {
            random = {
                chance = 5
                if = {
                    NOT = { has_trait = mute }
                }
                add_trait = mute
            }
            random = {
                chance = 5
                if = {
                    NOT = { has_trait = deaf }
                }
                add_trait = deaf
            }
            random = {
                chance = 10
                if = {
                    NOR = {
                        has_trait = hygene_bad_1
                        has_trait = hygene_bad_2
                        has_trait = hygene_good_1
                        has_trait = hygene_good_2
                    }
                }
                add_trait = hygene_good_1
            }
            random = {
                chance = 5
                if = {
                    NOR = {
                        has_trait = hygene_bad_1
                        has_trait = hygene_bad_2
                        has_trait = hygene_good_1
                        has_trait = hygene_good_2
                    }
                }
                add_trait = hygene_good_2
            }
            random = {
                chance = 10
                if = {
                    NOR = {
                        has_trait = hygene_bad_1
                        has_trait = hygene_bad_2
                        has_trait = hygene_good_1
                        has_trait = hygene_good_2
                    }
                }
                add_trait = hygene_bad_1
            }
            random = {
                chance = 8
                if = {
                    NOR = {
                        has_trait = hygene_bad_1
                        has_trait = hygene_bad_2
                        has_trait = hygene_good_1
                        has_trait = hygene_good_2
                    }
                }
                add_trait = hygene_bad_2
            }
        }
    }
}

tr_dwemer_buildings = {
    effect = {
        every_culture_county = {
            limit = {
                NOT = { has_holding_type = dwemer_forge_1 }
                culture = dwemer
            }
            set_holding_type = dwemer_forge_1
        }
    }
}

tr_fantasy_piety_gain = {
    effect = {
        tr_fantasy_piety_gain_change = yes
    }
}
