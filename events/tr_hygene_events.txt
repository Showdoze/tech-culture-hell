namespace = tr_hygene

tr_hygene.01 = { #Hygene Root
    type = character_event
    title = tr_hygene_event
    desc = tr_hygene_event_desc
	theme = mental_health
	window = visit_settlement_window

	left_portrait = {
		character = root
	}

    weight_multiplier = {
	    base = 0.2
		modifier = {
			add = 0.2
			OR = {
				has_trait = hygene_good_1
				has_trait = hygene_good_2
				has_trait = hygene_bad_1
				has_trait = hygene_bad_2
			}
		}
    }

	option = { # improve hygene
		name = tr_hygene_improve
		custom_tooltip = tr_hygene_improve_tip

		trigger = {
			NOT = {
				has_trait = hygene_good_2
			}
		}

		if = {
			limit = { 
				has_trait = hygene_good_1 
			}
			remove_trait = hygene_good_1
			remove_character_modifier = stress_smelling_of_sweat
			add_trait = hygene_good_2
		}
		else_if = {
			limit = { 
				has_trait = hygene_bad_1 
			}
			remove_trait = hygene_bad_1
		}
		else_if = {
			limit = { 
				has_trait = hygene_bad_2 
			}
			remove_trait = hygene_bad_2
			add_trait = hygene_bad_1
		}
		else = {
			add_trait = hygene_good_1
			remove_character_modifier = stress_smelling_of_sweat
		}

		ai_chance = {
			base = 5
			modifier = {
                add = -4
                has_trait = lazy
            }
			modifier = {
                add = 5
                has_trait = diligent
            }
			modifier = {
				faith = {
					OR = {
					    has_doctrine = special_doctrine_is_islamic_faith
					    has_doctrine = tenet_kemetic
						has_doctrine = tenet_atenist
						has_doctrine = tenet_sumerian
						has_doctrine = tenet_akkadian
						has_doctrine = tenet_babylonian
						has_doctrine = tenet_assyrian
						has_doctrine = tenet_philistinian
						has_doctrine = tenet_phoenician
						has_doctrine = tenet_punic
						has_doctrine = tenet_islamic
					}
				}
				add = 100
			}
			modifier = {
				culture = {
					OR = {
					    has_cultural_tradition = tradition_ce1_ritual_washing 
						has_cultural_tradition = tradition_egypt
					}
			    }	
				add = 100
			}
		}

		stress_impact = {
			base = 10
			lazy = 20
			diligent = -10
			stubborn = 10
			fickle = -10
			depressed_1 = 20
			gallant = -20
		}
	}

	option = { # keep hygene
		name = tr_hygene_keep
		custom_tooltip = tr_hygene_keep_tip

		ai_chance = {
			base = 10
			modifier = {
                add = 5
                has_trait = content
            }
		}

		stress_impact = {
			lazy = 10
			diligent = -10
			fickle = 10
			depressed_1 = 10
		}
	}

	option = { # decrease hygene
		name = tr_hygene_decrease
		custom_tooltip = tr_hygene_decrease_tip

		trigger = {
			NOT = {
				has_trait = hygene_bad_2
			}
		}

		if = {
			limit = { 
				has_trait = hygene_bad_1 
			}
			remove_trait = hygene_bad_1
			add_trait = hygene_bad_2
		}
		else_if = {
			limit = { 
				has_trait = hygene_good_1 
			}
			remove_trait = hygene_good_1
		}
		else_if = {
			limit = { 
				has_trait = hygene_good_2 
			}
			remove_trait = hygene_good_2
			add_trait = hygene_good_1
		}
		else = {
			add_trait = hygene_bad_1
		}

		ai_chance = {
			base = 4
			modifier = {
                add = 4
                has_trait = lazy
            }
			modifier = {
                add = -10
                has_trait = diligent
            }
			modifier = {
				culture = {
					has_cultural_pillar = heritage_indo_aryan
				}
				add = 50
			}
		}

		stress_impact = {
			base = -10
			diligent = 20
			lazy = -30
			stubborn = 10
			fickle = -10
			depressed_1 = -20
			gallant = 30
		}
	}
}
